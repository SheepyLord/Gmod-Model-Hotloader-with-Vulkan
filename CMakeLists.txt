# Vulkan compute solver. glslang, built from the pinned source as a host tool,
# compiles the kernel to SPIR-V that is embedded in the runtime; Vulkan itself
# is loaded at run time (vulkan-1.dll), so nothing links against the loader.
foreach(option GLSLANG_TESTS GLSLANG_ENABLE_INSTALL ENABLE_HLSL ENABLE_OPT ENABLE_GLSLANG_JS BUILD_EXTERNAL ENABLE_PCH)
  set(${option} OFF CACHE BOOL "" FORCE)
endforeach()
set(ENABLE_GLSLANG_BINARIES ON CACHE BOOL "" FORCE)
add_subdirectory(vendor/glslang EXCLUDE_FROM_ALL)
set(MMDHL_VULKAN_SPV "${CMAKE_BINARY_DIR}/generated/vulkan_solver_spv.h")
add_custom_command(OUTPUT "${MMDHL_VULKAN_SPV}"
  COMMAND glslang-standalone -V --target-env vulkan1.1 --vn mmdhl_vulkan_solver_spv -o "${MMDHL_VULKAN_SPV}" "${CMAKE_SOURCE_DIR}/shaders/vulkan/solver.comp"
  DEPENDS "${CMAKE_SOURCE_DIR}/shaders/vulkan/solver.comp" glslang-standalone VERBATIM)
# The profiling variant (MMDHL_VULKAN_CLOCK=1) records per-level SM clock cycles.
set(MMDHL_VULKAN_PROFILE_SPV "${CMAKE_BINARY_DIR}/generated/vulkan_solver_profile_spv.h")
add_custom_command(OUTPUT "${MMDHL_VULKAN_PROFILE_SPV}"
  COMMAND glslang-standalone -V --target-env vulkan1.1 -DMMDHL_PROFILE --vn mmdhl_vulkan_solver_profile_spv -o "${MMDHL_VULKAN_PROFILE_SPV}" "${CMAKE_SOURCE_DIR}/shaders/vulkan/solver.comp"
  DEPENDS "${CMAKE_SOURCE_DIR}/shaders/vulkan/solver.comp" glslang-standalone VERBATIM)
target_sources(mmd_runtime PRIVATE native/vulkan_solver.cpp "${MMDHL_VULKAN_SPV}" "${MMDHL_VULKAN_PROFILE_SPV}")
target_include_directories(mmd_runtime PRIVATE vendor/vulkan_headers/include)
