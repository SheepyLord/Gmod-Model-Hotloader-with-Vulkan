else()
  # GCC on Linux. Every binary carries its own libstdc++ (the game's Steam runtime
  # has an older one) and keeps its symbols private; the runtime's own API stays
  # exported, and each module binds its own definitions first, as DLLs do.
  set(CMAKE_POSITION_INDEPENDENT_CODE ON)
  set(MMDHL_WARNINGS -Wall -Wno-sign-compare -Wno-unused-variable -Wno-unused-but-set-variable -Wno-unused-function -Wno-parentheses -Wno-misleading-indentation)
  set(MMDHL_UTF8 "")
  set(MMDHL_DEBUG_COMPILE -g)
  set(MMDHL_DEBUG_LINK "")
  if(MMDHL_PLATFORM STREQUAL "linux")
    # SSE2 floating point, as on x86-64: x87 extended precision would change physics results.
    add_compile_options(-msse2 -mfpmath=sse)
    # Thread-local variables at fixed offsets from the thread pointer. GCC assumes the 32-bit
    # dynamic TLS call (___tls_get_addr) keeps SSE registers, but before glibc 2.39 its first
    # call in each thread allocates through an SSE memset and clobbers them (glibc bug
    # 31372): values held in them came back as garbage. The binaries' TLS is small.
    add_compile_options(-ftls-model=initial-exec)
  endif()
  # Unused code goes, with what only it referenced: the SDK headers' inline ConVar helpers
  # would otherwise need tier1 symbols the game does not export.
  add_compile_options(-ffunction-sections -fdata-sections)
  # libstdc++ is built in (the game's runtime has an older one); the unwinder stays the
  # shared libgcc_s, which every runtime has: exceptions cross from the runtime library to
  # the modules and the worker, and separate static unwinders abort on that.
  add_link_options(-static-libstdc++ -Wl,--exclude-libs,ALL -Wl,--disable-new-dtags -Wl,-z,origin -Wl,--gc-sections)
  set(CMAKE_BUILD_RPATH_USE_ORIGIN ON)
  set(CMAKE_SKIP_BUILD_RPATH OFF)
  set(CMAKE_BUILD_WITH_INSTALL_RPATH ON)
  set(CMAKE_INSTALL_RPATH "$ORIGIN")
  # Before glibc 2.34 (the release build's is 2.28) threads and dlopen are in libpthread
  # and libdl: every program and library links them.
  set(THREADS_PREFER_PTHREAD_FLAG ON)
  find_package(Threads REQUIRED)
  link_libraries(Threads::Threads ${CMAKE_DL_LIBS})
endif()
